Slicing into chunks
-------------------

def split_into_slices(sequence, slice_size):
    for i in range(0, len(sequence), slice_size):
        yield slice(i, i + slice_size)

a = np.arange(100)
my_slices = split_into_slices(a, 25)
for my_slice in my_slices:
    print(a[my_slice])
